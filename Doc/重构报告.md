```c
typedef struct {
    uint8_t  ParkStatus;      /* 0=空闲, 1=有车, 2=异常 */
    uint8_t  GeoMagnetic;     /* 0/1 */

    char     DeviceName[33];
    char     DeviceSecret[33];
    char     Version[24];

/* 命令回调函数类型 */
```

```c
void LoRa_Task(void)
{
    /* 响应网关轮询: 收到AT+DATA1 → 发送NodeData结构体 */
    LoRa_Node_Poll(LoRa_CmdCallback);

    /* 状态变化时主动标记(等下次网关轮询时发送) */
    /* 不主动发送，避免LoRa碰撞(网关轮询模式) */
}

static void LoRa_CmdCallback(const char *cmd, const char *value)
{
    if (strcmp(cmd, "AT+DATA1") == 0)
    {
        /* 网关查询数据 → 打包发送 */
        PackNodeData();
        LoRa_Node_SendData(&NodeDataCache);
        LoRaOnline = 1;
    }
    else if (strcmp(cmd, "AT+CER1") == 0)
    {
        /* 网关查询证书 → 发送证书 */
        LoRa_Node_SendCert(&NodeCert);
    }
    else if (strcmp(cmd, "AT+LedEnable") == 0)
    {
        /* 网关下发LED控制 */
        LEDEnable = (uint8_t)atoi(value);
        StatusChanged = 1;
    }
}
```
