/****************************************************************************
 * ESP8266 WiFi模块驱动 - esp8266.c
 *
 * 功能描述:
 *   实现ESP8266 WiFi模块的初始化、数据收发、AT指令解析等功能
 *   通过USART2与ESP8266通信
 *   支持WiFi连接、TCP连接、数据透传等功能
 *
 * 硬件配置:
 *   - USART2: TX-PA2, RX-PA3, 波特率115200
 *   - ESP8266模块: WiFi连接、TCP透传模式
 *
 * 缓冲区机制:
 *   - 环形缓冲区: 用于中断接收数据,防止数据丢失
 *   - 线性缓冲区: 用于数据处理和解析
 *
 * 作者: Bjmyhc
 * 日期: 2026-07-19
 ****************************************************************************/

#include "stm32f10x.h"
#include "esp8266.h"
#include "bsp_delay.h"
#include "bsp_usart.h"
#include <string.h>
#include <stdio.h>
#include "device_config.h"

/* WiFi账号和密码(引用device_config.h中的配置) */
#define ESP8266_WIFI_INFO     "AT+CWJAP=\"" WIFI_SSID "\",\"" WIFI_PWD "\"\r\n"

/* OneNET平台TCP连接信息 */
#define ESP8266_ONENET_INFO   "AT+CIPSTART=\"TCP\",\"mqtts.heclouds.com\",1883\r\n"

/* 环形缓冲区 */
static unsigned char esp8266_buf[ESP8266_BUF_SIZE];
static ESP8266_RxBuffer_t esp8266_rx = {0, 0, 0};

/* 线性缓冲区（用于数据处理） */
static unsigned char esp8266_lineBuf[ESP8266_BUF_SIZE];
static uint16_t esp8266_lineLen = 0;

/* 上次接收时间（用于判断接收完成） */
static uint32_t lastRxTick = 0;
#define RX_TIMEOUT_MS    100   /* 判断接收完成的兜底超时(ms)
                                 * 用于: 非透传(AT指令)模式 和
                                 *       透传模式下非MQTT杂讯的兜底判定 */
#define MQTT_PACKET_WAIT_MS  1000  /* 透传模式下等待MQTT报文凑满的最大时间(ms)
                                     * LoRa分包间隔通常<1s, 1s上限足够覆盖;
                                     * 超过说明链路丢包或收到非MQTT杂讯,
                                     * 返回当前数据让上层处理/清空, 避免半包
                                     * 永久滞留缓冲区阻塞后续报文接收 */

/* 透传模式标志: 1=已进入透传(AT+CIPSEND), 0=AT指令模式
 * 透传模式下:
 *   - SendData 直接发送裸数据, 不需要 AT+CIPSEND, 不清缓冲
 *   - GetIPD 返回裸MQTT字节, 不再解析 +IPD 头
 *   - AT指令不生效, 需先 ExitTransparent 退出 */
static uint8_t esp8266_inTransparent = 0;

/****************************************************************************
 * 函数名: ESP8266_Clear
 * 功能:   清空ESP8266接收缓冲区
 * 参数:   无
 * 返回值: 无
 * 说明:   清空环形缓冲区和线性缓冲区,包括头指针、尾指针、溢出标志
 ****************************************************************************/
void ESP8266_Clear(void)
{
    __disable_irq();
    esp8266_rx.head = 0;
    esp8266_rx.tail = 0;
    esp8266_rx.overflow = 0;
    esp8266_lineLen = 0;
    memset(esp8266_lineBuf, 0, sizeof(esp8266_lineBuf));
    __enable_irq();
}

/****************************************************************************
 * Function: ESP8266_Consume
 * Purpose:  Pop len bytes from the ring buffer (advance tail only).
 *           Unlike Clear(), it keeps the remaining bytes so a later
 *           GetIPD can return them. Enables incremental parsing when the
 *           buffer contains several MQTT packets / garbage + a real command.
 * Param:    len - number of bytes to drop
 * Return:   none
 ****************************************************************************/
void ESP8266_Consume(uint16_t len)
{
    uint16_t total = ESP8266_GetDataLen();
    __disable_irq();
    if (len > total)
        len = total;
    esp8266_rx.tail = (esp8266_rx.tail + len) % ESP8266_BUF_SIZE;
    __enable_irq();
}

/****************************************************************************
 * 函数名: ESP8266_GetDataLen
 * 功能:   获取环形缓冲区中数据长度
 * 参数:   无
 * 返回值: 当前缓冲区中数据字节数
 * 说明:   支持环形缓冲区的正常和回绕两种情况
 ****************************************************************************/
uint16_t ESP8266_GetDataLen(void)
{
    uint16_t len;
    __disable_irq();
    if (esp8266_rx.head >= esp8266_rx.tail)
        len = esp8266_rx.head - esp8266_rx.tail;
    else
        len = ESP8266_BUF_SIZE - esp8266_rx.tail + esp8266_rx.head;
    __enable_irq();
    return len;
}

/****************************************************************************
 * 函数名: ESP8266_IsOverflow
 * 功能:   检查缓冲区是否溢出
 * 参数:   无
 * 返回值: 1-溢出, 0-正常
 * 说明:   当缓冲区满时,新数据会被丢弃并设置溢出标志
 ****************************************************************************/
uint8_t ESP8266_IsOverflow(void)
{
    return esp8266_rx.overflow;
}

/****************************************************************************
 * 函数名: ESP8266_ClearOverflow
 * 功能:   清除缓冲区溢出标志
 * 参数:   无
 * 返回值: 无
 * 说明:   在处理完溢出情况后调用此函数清除标志
 ****************************************************************************/
void ESP8266_ClearOverflow(void)
{
    esp8266_rx.overflow = 0;
}

/****************************************************************************
 * 函数名: ESP8266_ReadByte
 * 功能:   从环形缓冲区读取一个字节
 * 参数:   无
 * 返回值: 读取的字节值, -1表示缓冲区为空
 * 说明:   内部函数,需保证在临界区调用
 ****************************************************************************/
static int16_t ESP8266_ReadByte(void)
{
    uint16_t data;

    if (esp8266_rx.head == esp8266_rx.tail)
        return -1;

    __disable_irq();
    data = esp8266_buf[esp8266_rx.tail];
    esp8266_rx.tail = (esp8266_rx.tail + 1) % ESP8266_BUF_SIZE;
    __enable_irq();

    return data;
}

/****************************************************************************
 * 函数名: ESP8266_MQTTFullLen
 * 功能:   从环形缓冲区头部解析并计算一个完整MQTT报文的长度(含固定头)
 * 参数:   无
 * 返回值: >0 - 完整MQTT报文的总字节数
 *         -1 - 报文长度字段未凑齐(数据不完整,需继续等待)
 *         0  - 非MQTT报文(杂讯或TCP残留)
 * 说明:   MQTT报文固定头格式:
 *           byte1: 报文类型(高4位) + 标志位(低4位)
 *           byte2..n: 剩余长度(Remaining Length)变长编码,
 *                     每字节低7位为有效数据, 最高位为"继续标志",
 *                     最多4个字节
 *         报文总长度 = 1(类型字节) + N(剩余长度编码字节) + 剩余长度
 *         在LoRa分片链路下意义重大: 必须等齐完整报文才能解析,
 *         否则半包会被误当作完整命令处理, 导致命令丢失
 ****************************************************************************/
static int16_t ESP8266_MQTTFullLen(void)
{
    uint16_t len = ESP8266_GetDataLen();
    uint16_t idx;
    uint8_t i;
    uint8_t j;
    uint8_t hdr[5];
    uint32_t remainLen = 0;
    uint32_t multiplier = 1;

    /* 报文未凑齐: 至少需要 类型字节 + 长度字节1个 */
    if (len < 2)
        return -1;

    /* peek 环形缓冲头部字节(不移动指针): hdr[0] = 类型字节,
     * hdr[1..] = 剩余长度(Remaining Length)变长编码字节 */
    for (i = 0; i < 5 && i < len; i++)
    {
        idx = (uint16_t)((esp8266_rx.tail + i) % ESP8266_BUF_SIZE);
        hdr[i] = esp8266_buf[idx];
    }
    /* 关键修复: 从 hdr[1] 开始解析剩余长度(之前误从 hdr[0] 解析,
     * 类型字节 bit7=0 直接 break, fullLen 恒为1, WaitRecive
     * 收到1字节就放行, LoRa分包永远未被拼装, 命令全丢
     * 正是命令丢失的根源 */
    for (j = 1; j < 5 && j < len; j++)
    {
        if (!(hdr[j] & 0x80))
            break;                      /* 剩余长度编码结束 */
    }
    /* 长度编码超过4个字节(异常) */
    if (j >= 5)
        return 0;

    /* 剩余长度字段尚未全部到达(需继续等待) */
    if (len < (uint16_t)(j + 1))
        return -1;

    /* 计算剩余长度 (hdr[1]..hdr[j]) */
    for (i = 1; i <= j; i++)
    {
        remainLen += (uint32_t)(hdr[i] & 0x7F) * multiplier;
        multiplier <<= 7;
    }

    /* 剩余长度超过缓冲区(不可能完整接收): 可能是非MQTT杂讯(如TCP残留),
     * 按非MQTT处理, 避免 int16_t 溢出 */
    if (remainLen > ESP8266_BUF_SIZE)
        return 0;

    /* 报文总长度 = 1(类型) + j(剩余长度编码字节) + remainLen */
    return (int16_t)(1 + j + remainLen);
}

/****************************************************************************
 * 函数名: ESP8266_GetLastRxTime
 * 功能:   获取最近一次有效接收的时间戳
 * 参数:   无
 * 返回值: 最近一次有效接收的时间(ms)
 * 说明:   用于链路活性检测: 若长时间(超过3倍心跳周期)
 *         没有任何有效接收(含PINGRESP), 判定
 *         LoRa/TCP链路已断开, 触发强制重连
 ****************************************************************************/
uint32_t ESP8266_GetLastRxTime(void)
{
    return lastRxTick;
}

/****************************************************************************
 * 函数名: ESP8266_WaitRecive
 * 功能:   判断数据是否接收完成
 * 参数:   无
 * 返回值: REV_OK-接收完成, REV_WAIT-接收未完成
 * 说明:   透传模式(MQTT报文): 按完整MQTT报文长度判定接收完成,
 *         报文未凑齐则继续等待, 防止LoRa分片报文被拆分处理;
 *         AT指令模式/非MQTT杂讯: 用超时兜底判定(10ms轮询判定)
 ****************************************************************************/
_Bool ESP8266_WaitRecive(void)
{
    uint32_t checkTick = Get_Tick();

    if (ESP8266_GetDataLen() == 0)
        return REV_WAIT;

    /* 透传模式: 按MQTT报文长度判定接收完成 */
    if (esp8266_inTransparent)
    {
        int16_t fullLen = ESP8266_MQTTFullLen();

        /* fullLen > 0 且 缓冲数据 >= 报文长度: 报文凑齐, 接收完成 */
        if (fullLen > 0 && ESP8266_GetDataLen() >= (uint16_t)fullLen)
        {
            lastRxTick = checkTick;
            return REV_OK;
        }
        /* fullLen <= 0: 非MQTT报文(如TCP残留), 用超时兜底 */
        if (fullLen <= 0)
        {
            if (checkTick - lastRxTick >= RX_TIMEOUT_MS)
            {
                lastRxTick = checkTick;
                return REV_OK;
            }
            return REV_WAIT;
        }
        /* fullLen > 0 但未凑齐: 报文不完整, 继续等待.
         * 若 MQTT_PACKET_WAIT_MS 超时: 判定LoRa链路切碎报文
         * 无法凑齐, 丢弃半包. 半包不处理, 不会误触发JSON parse error
         * 也不会回复错误400. 超时清理(即Drop partial)
         * 在下一层完成处理 */
        if (checkTick - lastRxTick >= MQTT_PACKET_WAIT_MS)
        {
            /* 不刷新lastRxTick: Drop partial是碎片清理, 不算有效
             * 数据接收. 若刷新lastRxTick, 即使TCP已断开, 碎片
             * 触发Drop partial, LINK_DEAD_TIMEOUT_MS永远无法触发,
             * 设备会误判链路存活持续假在线 */
            ESP8266_Clear();
            Usart_Printf(USART_DEBUG, "Drop partial MQTT packet (timeout)\r\n");
            return REV_WAIT;
        }
        return REV_WAIT;
    }

    /* 非透传(AT指令)模式: 用超时兜底判定 */
    if (checkTick - lastRxTick >= RX_TIMEOUT_MS)
    {
        lastRxTick = checkTick;
        return REV_OK;
    }

    return REV_WAIT;
}

/****************************************************************************
 * 函数名: ESP8266_CopyToLineBuf
 * 功能:   将环形缓冲区数据搬移到线性缓冲区
 * 参数:   无
 * 返回值: 搬移的字节数
 * 说明:   从环形缓冲区头部逐个读出所有字节, 搬运到线性缓冲区
 ****************************************************************************/
static uint16_t ESP8266_CopyToLineBuf(void)
{
    uint16_t len = ESP8266_GetDataLen();
    uint16_t i;
    int16_t data;

    esp8266_lineLen = 0;

    for (i = 0; i < len && i < ESP8266_BUF_SIZE - 1; i++)
    {
        data = ESP8266_ReadByte();
        if (data < 0)
            break;
        esp8266_lineBuf[esp8266_lineLen++] = (uint8_t)data;
    }
    esp8266_lineBuf[esp8266_lineLen] = '\0';

    return esp8266_lineLen;
}

/****************************************************************************
 * 函数名: ESP8266_CopyToLineBufPeek
 * 功能:   从环形缓冲区头部 peek(不移动指针) 复制 n 个字节到线性缓冲区
 * 参数:   n - 期望复制的字节数(不超过缓冲区现有数据量)
 * 返回值: 实际复制的字节数
 * 说明:   不推进tail, 环形缓冲数据保留, 之后由
 *         ESP8266_Consume 精确弹出已处理字节. 配合"消费式解析":
 *         GetIPD 每次只取一个完整报文, 剩余数据不丢失,
 *         "杂讯/伪报文 + 真实命令"同批到达时命令不会丢
 ****************************************************************************/
static uint16_t ESP8266_CopyToLineBufPeek(uint16_t n)
{
    uint16_t len = ESP8266_GetDataLen();
    uint16_t i;

    if (n > len)
        n = len;
    if (n > ESP8266_BUF_SIZE - 1)
        n = ESP8266_BUF_SIZE - 1;

    esp8266_lineLen = 0;
    for (i = 0; i < n; i++)
    {
        esp8266_lineBuf[esp8266_lineLen++] =
            esp8266_buf[(esp8266_rx.tail + i) % ESP8266_BUF_SIZE];
    }
    esp8266_lineBuf[esp8266_lineLen] = '\0';

    return esp8266_lineLen;
}

/****************************************************************************
 * 函数名: ESP8266_SendCmd
 * 功能:   发送AT命令并等待响应
 * 参数:   cmd - 要发送的AT命令
 *         res - 期望的响应字符串
 * 返回值: 0-成功(收到期望响应), 1-失败(超时未收到期望响应)
 * 说明:   最长等待约2秒(200次×10ms)
 ****************************************************************************/
_Bool ESP8266_SendCmd(char *cmd, char *res)
{
    unsigned char timeOut = 200;

    ESP8266_Clear();
    Usart_SendString(USART2, (unsigned char *)cmd, strlen((const char *)cmd));

    while (timeOut--)
    {
        if (ESP8266_WaitRecive() == REV_OK)
        {
            ESP8266_CopyToLineBuf();
            if (strstr((const char *)esp8266_lineBuf, res) != NULL)
            {
                ESP8266_Clear();
                return 0;
            }
        }

        DelayXms(10);
    }

    return 1;
}

/****************************************************************************
 * 函数名: ESP8266_SendData
 * 功能:   发送数据到ESP8266模块
 * 参数:   data - 数据指针
 *         len - 数据长度
 * 返回值: 无
 * 说明:   透传模式: 直接发送裸数据, 不需要AT+CIPSEND, 不清缓冲
 *         (注意: 透传时若先ESP8266_Clear会清掉下行数据)
 *         非透传模式: 先发送AT+CIPSEND指令(等待">")
 ****************************************************************************/
void ESP8266_SendData(unsigned char *data, unsigned short len)
{
    if (esp8266_inTransparent)
    {
        /* 透传模式: 直接发送, 不加AT+CIPSEND */
        Usart_SendString(USART2, data, len);
    }
    else
    {
        /* 非透传模式: 先发AT+CIPSEND指令 */
        char cmdBuf[32];
        ESP8266_Clear();
        sprintf(cmdBuf, "AT+CIPSEND=%d\r\n", len);
        if (!ESP8266_SendCmd(cmdBuf, ">"))
        {
            Usart_SendString(USART2, data, len);
        }
    }
}

/****************************************************************************
 * 函数名: ESP8266_GetIPD
 * 功能:   获取ESP8266接收到的数据(透传模式)
 * 参数:   timeOut - 超时参数(未使用, 兼容旧接口)
 * 返回值: 数据指针, NULL表示无数据
 * 说明:   透传模式: 返回完整MQTT报文(不解析+IPD头)
 *         非透传模式: 解析"+IPD,x:yyy"格式, 提取数据段
 *         配合peek+Consume, 每次只取一个报文
 ****************************************************************************/
unsigned char *ESP8266_GetIPD(unsigned short timeOut)
{
    char *ptrIPD = NULL;
    int16_t fullLen;

    if (ESP8266_WaitRecive() == REV_OK)
    {
        if (esp8266_inTransparent)
        {
            /* 透传模式: 返回完整MQTT报文(不解析+IPD头)
             * 关键改动: 用peek只取一个完整报文(MQTTFullLen计算长度),
             * 不推进tail, 剩余数据保留在环形缓冲. 由 OneNet_RevPro 尾部
             * ESP8266_Consume 精确弹出, 若LoRa切碎报文未凑齐
             * 数据不丢, 下轮继续等待拼装 */
            fullLen = ESP8266_MQTTFullLen();
            if (fullLen > 0)
            {
                ESP8266_CopyToLineBufPeek((uint16_t)fullLen);
            }
            else
            {
                /* fullLen<=0: 无法识别为MQTT报文(杂讯/残留), 全部peek
                 * 交给上层丢弃, 避免残留数据一直占用缓冲区 */
                ESP8266_CopyToLineBufPeek(ESP8266_GetDataLen());
            }
            if (esp8266_lineLen > 0)
            {
                return (unsigned char *)esp8266_lineBuf;
            }
        }
        else
        {
            /* 非透传模式: 解析+IPD头 */
            ESP8266_CopyToLineBuf();
            ptrIPD = strstr((char *)esp8266_lineBuf, "IPD,");
            if (ptrIPD != NULL)
            {
                ptrIPD = strchr(ptrIPD, ':');
                if (ptrIPD != NULL)
                {
                    ptrIPD++;
                    return (unsigned char *)(ptrIPD);
                }
            }
        }
    }

    return NULL;
}

/****************************************************************************
 * 函数名: ESP8266_Init
 * 功能:   初始化ESP8266 WiFi模块
 * 参数:   无
 * 返回值: 0-成功, 1-失败
 * 说明:   初始化步骤:
 *         1. AT指令 - 检测模块是否在线
 *         2. CWMODE - 设置STA模式
 *         3. CWDHCP - 使能DHCP
 *         4. CWJAP - 连接WiFi
 *         5. CIPCLOSE - 关闭遗留TCP连接
 *         6. CIPSTART - 连接OneNET服务器
 *         7. CIPMODE=1 - 开启透传模式
 *         8. CIPSEND - 进入透传, 开始发送MQTT数据
 *         每步最多重试3次, 失败立即返回(防止ESP8266掉电时卡死)
 ****************************************************************************/
_Bool ESP8266_Init(void)
{
    uint8_t retry;

    /* 复位透传标志, 防止上次状态残留
     * (重连时 Wifi_Reconnect 会先调用 ExitTransparent, 再重新初始化) */
    esp8266_inTransparent = 0;
    ESP8266_Clear();

    Usart_Printf(USART_DEBUG, "1. AT\r\n");
    for (retry = 0; retry < 3; retry++)
    {
        if (ESP8266_SendCmd("AT\r\n", "OK") == 0) break;
        DelayXms(500);
    }
    if (retry >= 3) return 1;

    Usart_Printf(USART_DEBUG, "2. CWMODE\r\n");
    for (retry = 0; retry < 3; retry++)
    {
        if (ESP8266_SendCmd("AT+CWMODE=1\r\n", "OK") == 0) break;
        DelayXms(500);
    }
    if (retry >= 3) return 1;

    Usart_Printf(USART_DEBUG, "3. AT+CWDHCP\r\n");
    for (retry = 0; retry < 3; retry++)
    {
        if (ESP8266_SendCmd("AT+CWDHCP=1,1\r\n", "OK") == 0) break;
        DelayXms(500);
    }
    if (retry >= 3) return 1;

    Usart_Printf(USART_DEBUG, "4. CWJAP\r\n");
    for (retry = 0; retry < 3; retry++)
    {
        if (ESP8266_SendCmd(ESP8266_WIFI_INFO, "GOT IP") == 0) break;
        DelayXms(500);
    }
    if (retry >= 3) return 1;

    /* 先关闭遗留的TCP连接, 防止CIPSTART后连接冲突
     * 或端口被占用(异常重连场景), 确保干净连接
     * 若没有遗留连接, CIPCLOSE失败不影响后续 */
    Usart_Printf(USART_DEBUG, "5. CIPCLOSE\r\n");
    Usart_SendString(USART2, (unsigned char *)"AT+CIPCLOSE\r\n", 13);
    DelayXms(300);
    ESP8266_Clear();

    Usart_Printf(USART_DEBUG, "6. CIPSTART\r\n");
    for (retry = 0; retry < 3; retry++)
    {
        if (ESP8266_SendCmd(ESP8266_ONENET_INFO, "CONNECT") == 0) break;
        DelayXms(500);
    }
    if (retry >= 3) return 1;

    /* 进入透传模式(CIPMODE=1 + CIPSEND)
     * 此后设备直接发MQTT裸数据, 不再使用AT+CIPSEND
     * 降低Lora传输开销(只传PUBLISH报文本身) */
    Usart_Printf(USART_DEBUG, "7. CIPMODE=1\r\n");
    for (retry = 0; retry < 3; retry++)
    {
        if (ESP8266_SendCmd("AT+CIPMODE=1\r\n", "OK") == 0) break;
        DelayXms(500);
    }
    if (retry >= 3) return 1;

    Usart_Printf(USART_DEBUG, "8. CIPSEND (enter transparent)\r\n");
    for (retry = 0; retry < 3; retry++)
    {
        if (ESP8266_SendCmd("AT+CIPSEND\r\n", ">") == 0) break;
        DelayXms(500);
    }
    if (retry >= 3) return 1;

    esp8266_inTransparent = 1;
    ESP8266_Clear();   /* 清掉进入透传前可能残留的杂讯 */
    Usart_Printf(USART_DEBUG, "9. ESP8266 Init OK (transparent mode)\r\n");
    return 0;
}

/****************************************************************************
 * 函数名: ESP8266_IsTransparent
 * 功能:   查询当前是否处于透传模式
 * 参数:   无
 * 返回值: 1-透传模式, 0-非透传(AT指令)模式
 ****************************************************************************/
uint8_t ESP8266_IsTransparent(void)
{
    return esp8266_inTransparent;
}

/****************************************************************************
 * 函数名: ESP8266_ExitTransparent
 * 功能:   退出透传模式, 恢复AT指令模式
 * 参数:   无
 * 返回值: 无
 * 说明:   发送"+++"退出指令(前面需空闲1秒以上)
 *         退出前需保证1秒无数据, 避免ESP8266误判
 *         退出后清空缓冲区, 后续可正常发送AT指令(CIPCLOSE/CIPSTART等)
 ****************************************************************************/
void ESP8266_ExitTransparent(void)
{
    uint8_t retry;

    if (!esp8266_inTransparent)
        return;

    /* 先标记非透传, 使WaitRecive走AT指令模式(超时判定)解析AT响应 */
    esp8266_inTransparent = 0;

    /* 重试3次: 发+++后用AT验证是否成功退出透传模式
     * 透传模式下+++可能被当作TCP数据吞掉, 不验证后续AT指令全部无效 */
    for (retry = 0; retry < 3; retry++)
    {
        /* +++前等待: 确保无数据干扰(断线场景500ms足够) */
        DelayXms(500);

        /* 发+++退出透传(裸发, 不加\r\n) */
        Usart_SendString(USART2, (unsigned char *)"+++", 3);
        DelayXms(500);

        /* AT验证: SendCmd内部清缓冲+发AT+等OK响应(超时2s) */
        if (ESP8266_SendCmd("AT\r\n", "OK") == 0)
        {
            Usart_Printf(USART_DEBUG, "Exit transparent OK\r\n");
            return;
        }

        Usart_Printf(USART_DEBUG, "Exit transparent retry %d\r\n", retry + 1);
    }

    /* 3次验证均失败: 仍继续, ESP8266_Init的AT会再试一次 */
    Usart_Printf(USART_DEBUG, "Exit transparent FAILED\r\n");
    ESP8266_Clear();
}

/****************************************************************************
 * 函数名: USART2_IRQHandler
 * 功能:   串口2接收中断处理函数
 * 参数:   无
 * 返回值: 无
 * 说明:   将接收到的数据写入环形缓冲区
 *         缓冲区满时丢弃新数据并置溢出标志
 ****************************************************************************/
void USART2_IRQHandler(void)
{
    if (USART_GetITStatus(USART2, USART_IT_RXNE) != RESET)
    {
        uint16_t nextHead;
        uint8_t data;

        data = USART2->DR;

        nextHead = (esp8266_rx.head + 1) % ESP8266_BUF_SIZE;

        if (nextHead == esp8266_rx.tail)
        {
            esp8266_rx.overflow = 1;
        }
        else
        {
            esp8266_buf[esp8266_rx.head] = data;
            esp8266_rx.head = nextHead;
            lastRxTick = Get_Tick();
        }

        USART_ClearFlag(USART2, USART_FLAG_RXNE);
    }
}
